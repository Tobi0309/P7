cmake_minimum_required(VERSION 3.16)
project(frontier_viewpoint_optimizer)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(octomap REQUIRED)
find_package(octomap_msgs REQUIRED)
find_package(frontier_segmentation REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Eigen3 REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/OptimizedFrontier.msg" "msg/OptimizedFrontierArray.msg"
  DEPENDENCIES std_msgs geometry_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_executable(frontier_viewpoint_optimizer_node src/frontier_viewpoint_optimizer_node.cpp)
target_include_directories(frontier_viewpoint_optimizer_node PRIVATE include ${OCTOMAP_INCLUDE_DIRS})
ament_target_dependencies(frontier_viewpoint_optimizer_node rclcpp geometry_msgs std_msgs
  visualization_msgs nav_msgs octomap octomap_msgs frontier_segmentation)
target_link_libraries(frontier_viewpoint_optimizer_node ${OCTOMAP_LIBRARIES}
  Eigen3::Eigen nlohmann_json::nlohmann_json "${cpp_typesupport_target}")
find_path(ROS2TOP_INCLUDE_DIR ros2top/ros2top.hpp)
if(ROS2TOP_INCLUDE_DIR)
  target_include_directories(frontier_viewpoint_optimizer_node PRIVATE ${ROS2TOP_INCLUDE_DIR})
  target_compile_definitions(frontier_viewpoint_optimizer_node PRIVATE FRONTIER_OPTIMIZER_HAS_ROS2TOP=1)
endif()
install(TARGETS frontier_viewpoint_optimizer_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
