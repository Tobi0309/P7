cmake_minimum_required(VERSION 3.8)
project(frontier_segmentation)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(octomap REQUIRED)
find_package(octomap_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Frontier.msg"
  "msg/FrontierArray.msg"
  DEPENDENCIES geometry_msgs std_msgs)
add_executable(frontier_segmentation_node src/frontier_segmentation_node.cpp)
ament_target_dependencies(frontier_segmentation_node rclcpp octomap octomap_msgs geometry_msgs std_msgs visualization_msgs tf2 tf2_ros)
target_include_directories(frontier_segmentation_node PRIVATE ${OCTOMAP_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
target_link_libraries(frontier_segmentation_node ${OCTOMAP_LIBRARIES})
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(frontier_segmentation_node "${cpp_typesupport_target}")
install(TARGETS frontier_segmentation_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
